
# 学校管理系统
## 介绍
依托校园的管理，所构建的一个前后端分离，全局使用CRUD模块的异步交流的教师学生双系统智能安全Web工作平台。
## 集成配置
```
spring:
  datasource:
    url: jdbc:mysql://localhost:3306/student_db?useSSL=false&serverTimezone=Asia/Shanghai
    username: root
    password: csr20051122
    driver-class-name: com.mysql.cj.jdbc.Driver
  # Redis
  data:
    redis:
      host: localhost
      port: 6379
      password:
      timeout: 5000ms
      lettuce:
        pool:
          max-active: 8
          max-idle: 8
          min-idle: 0
mybatis:
  mapper-locations: classpath:mapper/*.xml
  type-aliases-package: com.test.student.entity

logging:
  level:
    com.test.student.mapper: DEBUG
```

### 3.运用AJAX，使前后端分离
#### 如
```
loginForm.addEventListener('submit',async function (e){
            console.log("have")
            e.preventDefault();
            const params = new URLSearchParams();
            params.append('username', usernameInput.value.trim());
            params.append('password', passwordInput.value);
            params.append('roleType', roleType)
            const response =await fetch('/doLogin',{
                method:'POST',
                headers:{
                    'Content-Type': 'application/x-www-form-urlencoded'
                },
                body:params
            })
            const res=await response.json();
            console.log(res)
            console.log(res.data)
            if (res.data){
                console.log('准备跳转')
                window.location.href=res.data;
            }
        })
```
